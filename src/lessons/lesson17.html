Урок 17: Робота з формами через JS

Отримання даних, валідація, FormData, динамічні форми

Основи роботи з формами

Форми — це основний спосіб отримання даних від користувача. JavaScript дозволяє отримувати значення полів, валідувати дані перед відправкою та створювати динамічні форми.

Основні задачі при роботі з формами

  • Отримання значень з полів форми
  • Валідація даних на стороні клієнта
  • Запобігання відправки форми (preventDefault)
  • Динамічне додавання/видалення полів
  • Робота з FormData API
  • Відображення помилок валідації

Обробка відправки форми

const form = document.querySelector('#myForm');

// Подія submit спрацьовує при відправці форми
form.addEventListener('submit', (e) => {
    // Запобігти стандартній відправці форми
    e.preventDefault();
    
    // Отримати дані
    const formData = {
        name: form.elements.name.value,
        email: form.elements.email.value,
        message: form.elements.message.value
    };
    
    console.log('Дані форми:', formData);
    
    // Тут можна відправити дані на сервер
    // fetch('/api/submit', { method: 'POST', body: JSON.stringify(formData) })
});

// Програмна відправка форми
form.submit();  // Викликає submit без перевірок

// Програмне натискання кнопки submit
const submitButton = form.querySelector('button[type="submit"]');
submitButton.click();  // Викликає submit з перевірками

// Скидання форми
form.reset();  // Очищає всі поля

// Приклад: повна обробка форми
form.addEventListener('submit', async (e) => {
    e.preventDefault();
    
    // Отримати дані
    const data = new FormData(form);
    
    // Показати індикатор завантаження
    const submitBtn = form.querySelector('button[type="submit"]');
    submitBtn.disabled = true;
    submitBtn.textContent = 'Відправка...';
    
    try {
        // Відправити дані
        const response = await fetch('/api/submit', {
            method: 'POST',
            body: data
        });
        
        if (response.ok) {
            alert('Форма успішно відправлена!');
            form.reset();
        } else {
            alert('Помилка відправки');
        }
    } catch (error) {
        console.error('Помилка:', error);
        alert('Помилка з\'єднання');
    } finally {
        submitBtn.disabled = false;
        submitBtn.textContent = 'Відправити';
    }
});

Валідація форм

// HTML5 валідація (атрибути)
// required, minlength, maxlength, pattern, type="email", type="url" тощо

// JavaScript валідація
function validateForm(form) {
    const errors = [];
    
    // Перевірка імені
    const name = form.elements.name.value.trim();
    if (name.length < 2) {
        errors.push('Ім\'я повинно містити мінімум 2 символи');
    }
    
    // Перевірка email
    const email = form.elements.email.value.trim();
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email)) {
        errors.push('Некоректний email');
    }
    
    // Перевірка телефону
    const phone = form.elements.phone.value.trim();
    const phoneRegex = /^\+?[\d\s\-\(\)]+$/;
    if (phone && !phoneRegex.test(phone)) {
        errors.push('Некоректний номер телефону');
    }
    
    // Перевірка пароля
    const password = form.elements.password.value;
    if (password.length < 8) {
        errors.push('Пароль повинен містити мінімум 8 символів');
    }
    
    // Перевірка підтвердження пароля
    const passwordConfirm = form.elements.passwordConfirm.value;
    if (password !== passwordConfirm) {
        errors.push('Паролі не збігаються');
    }
    
    // Перевірка checkbox
    const agree = form.elements.agree.checked;
    if (!agree) {
        errors.push('Необхідно прийняти умови');
    }
    
    return errors;
}

// Використання
form.addEventListener('submit', (e) => {
    e.preventDefault();
    
    const errors = validateForm(form);
    
    if (errors.length > 0) {
        showErrors(errors);
    } else {
        // Відправити форму
        submitForm(form);
    }
});

// Показ помилок
function showErrors(errors) {
    const errorContainer = document.querySelector('.errors');
    errorContainer.innerHTML = errors
        .map(err => `
${err}
`)
        .join('');
}

// Валідація в реальному часі
const emailInput = form.elements.email;
emailInput.addEventListener('blur', () => {
    const email = emailInput.value.trim();
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    
    if (!emailRegex.test(email)) {
        emailInput.classList.add('invalid');
        showFieldError(emailInput, 'Некоректний email');
    } else {
        emailInput.classList.remove('invalid');
        hideFieldError(emailInput);
    }
});

// Constraint Validation API
const input = form.elements.email;

// Перевірка валідності
if (!input.validity.valid) {
    console.log('Поле не валідне');
    console.log(input.validationMessage);  // Повідомлення про помилку
}

// Власне повідомлення про помилку
input.setCustomValidity('Цей email вже використовується');

// Скасувати власне повідомлення
input.setCustomValidity('');

FormData API

// FormData - зручний спосіб збору даних форми
const form = document.querySelector('#myForm');

// Створення FormData з форми
const formData = new FormData(form);

// Або створення порожньої FormData
const data = new FormData();

// Додавання даних
data.append('name', 'Іван');
data.append('email', 'ivan@example.com');
data.append('age', 25);

// Додавання файлу
const fileInput = document.querySelector('#avatar');
data.append('avatar', fileInput.files[0]);

// Отримання значення
const name = data.get('name');  // 'Іван'

// Перевірка наявності поля
if (data.has('email')) {
    console.log('Email присутній');
}

// Видалення поля
data.delete('age');

// Встановлення значення (замінює існуюче)
data.set('name', 'Петро');

// Отримання всіх значень поля (для полів з однаковими іменами)
const hobbies = data.getAll('hobby');

// Перебір даних
for (let [key, value] of data.entries()) {
    console.log(key, value);
}

// Конвертація FormData в об'єкт
const obj = Object.fromEntries(data.entries());
console.log(obj);

// Відправка FormData через fetch
fetch('/api/submit', {
    method: 'POST',
    body: formData  // не потрібно встановлювати Content-Type
});

// FormData з файлами
form.addEventListener('submit', async (e) => {
    e.preventDefault();
    
    const formData = new FormData(form);
    
    // Додати додаткові дані
    formData.append('timestamp', Date.now());
    formData.append('userId', '123');
    
    try {
        const response = await fetch('/api/upload', {
            method: 'POST',
            body: formData
        });
        
        const result = await response.json();
        console.log('Результат:', result);
    } catch (error) {
        console.error('Помилка:', error);
    }
});

// Додавання полів динамічно
const addFieldButton = document.querySelector('#add-field');
const fieldsContainer = document.querySelector('#fields-container');

addFieldButton.addEventListener('click', () => {
    const fieldDiv = document.createElement('div');
    fieldDiv.className = 'field-group';
    fieldDiv.innerHTML = `
        
        
    `;
    
    fieldsContainer.appendChild(fieldDiv);
});

// Видалення полів
fieldsContainer.addEventListener('click', (e) => {
    if (e.target.classList.contains('remove-field')) {
        e.target.parentElement.remove();
    }
});

// Приклад: динамічний список завдань
const todoContainer = document.querySelector('#todo-list');
const addTodoBtn = document.querySelector('#add-todo');

let todoCount = 0;

addTodoBtn.addEventListener('click', () => {
    todoCount++;
    
    const todoDiv = document.createElement('div');
    todoDiv.className = 'todo-item';
    todoDiv.innerHTML = `
        
        
        
    `;
    
    todoContainer.appendChild(todoDiv);
});

// Умовні поля
const userTypeSelect = document.querySelector('#user-type');
const companyFields = document.querySelector('#company-fields');

userTypeSelect.addEventListener('change', (e) => {
    if (e.target.value === 'business') {
        companyFields.style.display = 'block';
        // Зробити поля обов'язковими
        companyFields.querySelectorAll('input').forEach(input => {
            input.required = true;
        });
    } else {
        companyFields.style.display = 'none';
        companyFields.querySelectorAll('input').forEach(input => {
            input.required = false;
        });
    }
});

// Автозаповнення
const cityInput = document.querySelector('#city');
const suggestions = document.querySelector('#city-suggestions');
const cities = ['Київ', 'Харків', 'Одеса', 'Дніпро', 'Львів'];

cityInput.addEventListener('input', (e) => {
    const value = e.target.value.toLowerCase();
    
    if (value.length > 0) {
        const matches = cities.filter(city => 
            city.toLowerCase().includes(value)
        );
        
        suggestions.innerHTML = matches
            .map(city => `
${city}
`)
            .join('');
        suggestions.style.display = 'block';
    } else {
        suggestions.style.display = 'none';
    }
});

suggestions.addEventListener('click', (e) => {
    if (e.target.classList.contains('suggestion')) {
        cityInput.value = e.target.textContent;
        suggestions.style.display = 'none';
    }
});
