Урок 13: Основи JavaScript

Змінні, типи даних, оператори та базовий синтаксис

Що таке JavaScript?

JavaScript - це мова програмування, яка дозволяє створювати інтерактивні веб-сторінки. Вона виконується в браузері користувача і може змінювати HTML, CSS, обробляти події та взаємодіяти з сервером.

Можливості JavaScript

  • Динамічна зміна контенту сторінки
  • Обробка подій користувача (клік, введення тексту тощо)
  • Валідація форм
  • Анімації та ефекти
  • Взаємодія з сервером (AJAX, Fetch API)
  • Зберігання даних локально

Змінні: let та const

Змінні дозволяють зберігати дані. В сучасному JavaScript використовуються let для змінних значень та const для констант. 

// const - для значень, які не змінюються
const PI = 3.14159;
const siteName = "Мій сайт";

// let - для значень, які можуть змінюватися
let age = 25;
let username = "Іван";
let isLoggedIn = false;

// Зміна значення
age = 26;  // OK
username = "Петро";  // OK

// PI = 3.14;  // Помилка! const не можна перевизначити

// Уникайте var - це застарілий спосіб
var oldWay = "не рекомендується";

Типи даних

JavaScript має кілька базових типів даних:

// Number - числа (цілі та дробові)
let integer = 42;
let decimal = 3.14;
let negative = -10;

// String - рядки (текст)
let text = "Привіт";
let text2 = 'Світ';
let text3 = `Це шаблонний рядок`;

// Boolean - логічний тип (true або false)
let isActive = true;
let hasError = false;

// Undefined - не визначено
let notDefined;
console.log(notDefined);  // undefined

// Null - пусте значення
let empty = null;

// Object - об'єкт
let person = {
    name: "Іван",
    age: 25,
    isStudent: true
};

// Array - масив
let numbers = [1, 2, 3, 4, 5];
let fruits = ["яблуко", "банан", "апельсин"];

// Перевірка типу
console.log(typeof 42);          // "number"
console.log(typeof "текст");     // "string"
console.log(typeof true);        // "boolean"
console.log(typeof {});          // "object"

Математичні оператори

// Базові операції
let a = 10;
let b = 3;

console.log(a + b);   // 13 - додавання
console.log(a - b);   // 7  - віднімання
console.log(a * b);   // 30 - множення
console.log(a / b);   // 3.333... - ділення
console.log(a % b);   // 1  - остача від ділення
console.log(a ** b);  // 1000 - піднесення до степеня

// Скорочені операції
let x = 10;
x += 5;  // x = x + 5;  результат: 15
x -= 3;  // x = x - 3;  результат: 12
x *= 2;  // x = x * 2;  результат: 24
x /= 4;  // x = x / 4;  результат: 6

// Інкремент та декремент
let count = 0;
count++;  // count = count + 1;  результат: 1
count--;  // count = count - 1;  результат: 0

// Префіксний та постфіксний інкремент
let y = 5;
console.log(y++);  // Виведе 5, потім y стане 6
console.log(++y);  // Спочатку y стане 7, потім виведе 7

Робота з рядками (Strings)

// Конкатенація (об'єднання) рядків
let firstName = "Іван";
let lastName = "Петренко";
let fullName = firstName + " " + lastName;
console.log(fullName);  // "Іван Петренко"

// Шаблонні рядки (template literals) - сучасний спосіб
let age = 25;
let greeting = `Привіт, мене звуть ${firstName} і мені ${age} років`;
console.log(greeting);

// Багаторядкові рядки
let poem = `Рядок перший
Рядок другий
Рядок третій`;

// Корисні методи рядків
let text = "JavaScript";

console.log(text.length);           // 10 - довжина рядка
console.log(text.toLowerCase());    // "javascript" - в нижній регістр
console.log(text.toUpperCase());    // "JAVASCRIPT" - в верхній регістр
console.log(text.includes("Script")); // true - чи містить підрядок
console.log(text.startsWith("Java")); // true - чи починається з
console.log(text.endsWith("Script")); // true - чи закінчується на
console.log(text.slice(0, 4));      // "Java" - вирізати частину
console.log(text.replace("Java", "Type")); // "TypeScript" - замінити

// Перетворення рядка в число
let strNumber = "123";
let num = Number(strNumber);  // 123
let num2 = parseInt("123");   // 123
let num3 = parseFloat("12.5"); // 12.5

// Перетворення числа в рядок
let number = 42;
let str = String(number);     // "42"
let str2 = number.toString(); // "42"

Оператори порівняння

// Строге порівняння (рекомендується)
console.log(5 === 5);      // true
console.log(5 === "5");    // false (різні типи)
console.log(5 !== 3);      // true
console.log(5 !== "5");    // true (різні типи)

// Нестроге порівняння (не рекомендується)
console.log(5 == "5");     // true (приводить типи)
console.log(5 != "3");     // true

// Порівняння чисел
console.log(10 > 5);       // true
console.log(10 < 5);       // false
console.log(10 >= 10);     // true
console.log(10 <= 9);      // false

// Логічні оператори
let age = 20;
let hasLicense = true;

// AND (&&) - всі умови мають бути true
console.log(age >= 18 && hasLicense);  // true

// OR (||) - хоча б одна умова має бути true
console.log(age < 18 || hasLicense);   // true

// NOT (!) - інвертує значення
console.log(!hasLicense);  // false
console.log(!(age < 18));  // true

Умовні конструкції

// if - else
let age = 20;

if (age >= 18) {
    console.log("Ви повнолітній");
} else {
    console.log("Ви неповнолітній");
}

// if - else if - else
let score = 85;

if (score >= 90) {
    console.log("Оцінка: Відмінно");
} else if (score >= 75) {
    console.log("Оцінка: Добре");
} else if (score >= 60) {
    console.log("Оцінка: Задовільно");
} else {
    console.log("Оцінка: Незадовільно");
}

// Тернарний оператор - коротка форма if-else
let isAdult = age >= 18 ? "так" : "ні";
console.log("Повнолітній:", isAdult);

let status = score >= 60 ? "Склав" : "Не склав";

// Switch - для множинного вибору
let day = 3;
let dayName;

switch(day) {
    case 1:
        dayName = "Понеділок";
        break;
    case 2:
        dayName = "Вівторок";
        break;
    case 3:
        dayName = "Середа";
        break;
    case 4:
        dayName = "Четвер";
        break;
    case 5:
        dayName = "П'ятниця";
        break;
    case 6:
    case 7:
        dayName = "Вихідний";
        break;
    default:
        dayName = "Невідомий день";
}

Цикли

// for - класичний цикл
for (let i = 0; i < 5; i++) {
    console.log("Ітерація:", i);
}

// while - виконується поки умова true
let count = 0;
while (count < 3) {
    console.log("Count:", count);
    count++;
}

// do-while - виконається мінімум один раз
let num = 0;
do {
    console.log("Number:", num);
    num++;
} while (num < 3);

// for...of - для масивів
let fruits = ["яблуко", "банан", "апельсин"];
for (let fruit of fruits) {
    console.log(fruit);
}

// for...in - для об'єктів
let person = { name: "Іван", age: 25, city: "Київ" };
for (let key in person) {
    console.log(key + ":", person[key]);
}

// break - вихід з циклу
for (let i = 0; i < 10; i++) {
    if (i === 5) break;
    console.log(i);  // 0, 1, 2, 3, 4
}

// continue - пропустити ітерацію
for (let i = 0; i < 5; i++) {
    if (i === 2) continue;
    console.log(i);  // 0, 1, 3, 4
}
