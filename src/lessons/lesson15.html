Урок 15: DOM маніпуляції

Взаємодія з HTML через JavaScript: querySelector, createElement, classList

Що таке DOM?

DOM (Document Object Model) — це програмний інтерфейс для HTML документів. Він представляє структуру сторінки у вигляді дерева об'єктів, які можна змінювати через JavaScript.

Можливості DOM маніпуляцій

  • Знаходити та отримувати доступ до елементів
  • Змінювати текст та HTML вміст
  • Додавати, видаляти та переміщувати елементи
  • Змінювати стилі та класи елементів
  • Отримувати та змінювати атрибути

Пошук елементів: querySelector

Сучасні методи querySelector та querySelectorAll використовують CSS-селектори для пошуку елементів. 

// querySelector - знаходить перший елемент
const title = document.querySelector('h1');
const firstButton = document.querySelector('.button');
const mainContent = document.querySelector('#main-content');

// querySelectorAll - знаходить всі елементи (повертає NodeList)
const allButtons = document.querySelectorAll('.button');
const allLinks = document.querySelectorAll('a');

// Складні селектори
const activeItem = document.querySelector('.item.active');
const firstListItem = document.querySelector('ul > li:first-child');
const checkedInputs = document.querySelectorAll('input:checked');

// Пошук усередині елемента
const container = document.querySelector('.container');
const childParagraph = container.querySelector('p');

// Старі методи (для довідки, краще використовувати querySelector)
const byId = document.getElementById('my-id');
const byClass = document.getElementsByClassName('my-class');
const byTag = document.getElementsByTagName('div');

Зміна вмісту елементів

// textContent - змінює тільки текст (безпечно)
const heading = document.querySelector('h1');
heading.textContent = 'Новий заголовок';

// innerHTML - змінює HTML (потенційно небезпечно з користувацькими даними)
const container = document.querySelector('.container');
container.innerHTML = '
Новий HTML вміст
';

// innerText - як textContent, але враховує CSS (видимість)
const paragraph = document.querySelector('p');
paragraph.innerText = 'Видимий текст';

// Приклад зміни вмісту
const counter = document.querySelector('#counter');
let count = 0;

function increment() {
    count++;
    counter.textContent = count;
}

// Додавання до існуючого вмісту
const list = document.querySelector('ul');
list.innerHTML += '
  • Новий елемент
    • ';

// Безпечніше додавати через += для textContent
const output = document.querySelector('.output');
output.textContent += 'Додатковий текст\n';

    Створення та додавання елементів

    // createElement - створює новий елемент
const newDiv = document.createElement('div');
const newParagraph = document.createElement('p');
const newButton = document.createElement('button');

// Налаштування створеного елемента
newParagraph.textContent = 'Це новий параграф';
newParagraph.className = 'text-block';
newParagraph.id = 'unique-id';

newButton.textContent = 'Натисни мене';
newButton.onclick = function() {
    alert('Клік!');
};

// Додавання елементів
const container = document.querySelector('.container');

// appendChild - додає в кінець
container.appendChild(newParagraph);

// append - може додати кілька елементів або текст
container.append(newButton, ' та текст');

// prepend - додає на початок
container.prepend(newDiv);

// insertBefore - вставляє перед вказаним елементом
const reference = document.querySelector('.reference');
container.insertBefore(newDiv, reference);

// insertAdjacentHTML - вставляє HTML
container.insertAdjacentHTML('beforeend', '
    HTML вміст
    ');
// 'beforebegin' - перед елементом
// 'afterbegin' - на початок елемента
// 'beforeend' - в кінець елемента
// 'afterend' - після елемента

// Створення списку динамічно
const ul = document.createElement('ul');
const items = ['Яблуко', 'Банан', 'Апельсин'];

items.forEach(item => {
    const li = document.createElement('li');
    li.textContent = item;
    ul.appendChild(li);
});

document.body.appendChild(ul);

    Видалення елементів

    // remove - видаляє сам елемент
const element = document.querySelector('.to-remove');
element.remove();

// removeChild - видаляє дочірній елемент
const parent = document.querySelector('.parent');
const child = document.querySelector('.child');
parent.removeChild(child);

// Видалення всіх дочірніх елементів
const container = document.querySelector('.container');
container.innerHTML = ''; // Швидкий спосіб

// Або через цикл
while (container.firstChild) {
    container.removeChild(container.firstChild);
}

// Видалення елемента після кліку
const buttons = document.querySelectorAll('.delete-btn');
buttons.forEach(btn => {
    btn.addEventListener('click', function() {
        this.parentElement.remove(); // видалити батьківський елемент
    });
});

    Робота з класами: classList

    const element = document.querySelector('.box');

// add - додати клас
element.classList.add('active');
element.classList.add('highlight', 'bordered'); // кілька класів

// remove - видалити клас
element.classList.remove('active');

// toggle - переключити клас (якщо є - видалити, якщо немає - додати)
element.classList.toggle('visible');

// contains - перевірити наявність класу
if (element.classList.contains('active')) {
    console.log('Елемент активний');
}

// replace - замінити один клас на інший
element.classList.replace('old-class', 'new-class');

// Приклад: перемикання теми
const themeToggle = document.querySelector('#theme-toggle');
themeToggle.addEventListener('click', () => {
    document.body.classList.toggle('dark-theme');
});

// Приклад: табуляція
const tabs = document.querySelectorAll('.tab');
tabs.forEach(tab => {
    tab.addEventListener('click', () => {
        // Видалити active з усіх
        tabs.forEach(t => t.classList.remove('active'));
        // Додати до поточного
        tab.classList.add('active');
    });
});

    Робота з атрибутами

    const link = document.querySelector('a');
const image = document.querySelector('img');
const input = document.querySelector('input');

// getAttribute - отримати значення атрибута
const href = link.getAttribute('href');
const src = image.getAttribute('src');

// setAttribute - встановити атрибут
link.setAttribute('href', 'https://example.com');
link.setAttribute('target', '_blank');
image.setAttribute('alt', 'Опис зображення');

// removeAttribute - видалити атрибут
link.removeAttribute('target');

// hasAttribute - перевірити наявність атрибута
if (link.hasAttribute('href')) {
    console.log('Посилання має атрибут href');
}

// Пряме звернення до атрибутів (для стандартних атрибутів)
link.href = 'https://example.com';
image.src = 'image.jpg';
input.value = 'текст';
input.disabled = true;

// data-атрибути
const element = document.querySelector('.user-card');
element.dataset.userId = '123';
element.dataset.userName = 'Іван';

console.log(element.dataset.userId);  // '123'
console.log(element.getAttribute('data-user-id'));  // '123'

    Зміна стилів через JavaScript

    const element = document.querySelector('.box');

// style - пряма зміна CSS властивостей
element.style.color = 'red';
element.style.backgroundColor = 'yellow';
element.style.fontSize = '20px';
element.style.padding = '10px';

// cssText - встановлення кількох стилів одразу
element.style.cssText = 'color: blue; background: white; padding: 15px;';

// Отримання значення стилю
console.log(element.style.color);  // 'blue'

// getComputedStyle - отримати фактичні обчислені стили (включно з CSS)
const computed = window.getComputedStyle(element);
console.log(computed.color);        // rgb(0, 0, 255)
console.log(computed.fontSize);     // 20px
console.log(computed.display);      // block

// Видалення стилю
element.style.color = '';  // видаляє inline стиль

// Приклад: зміна кольору при наведенні
const button = document.querySelector('button');
button.addEventListener('mouseenter', () => {
    button.style.backgroundColor = '#4CAF50';
});
button.addEventListener('mouseleave', () => {
    button.style.backgroundColor = '';
});
