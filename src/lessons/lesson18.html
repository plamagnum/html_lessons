Урок 18: LocalStorage та JSON

Збереження даних у браузері: localStorage, sessionStorage, JSON

Що таке Web Storage?

Web Storage API дозволяє зберігати дані безпосередньо в браузері користувача. Існує два типи сховищ: localStorage (постійне) та sessionStorage (тимчасове).

Основні відмінності Storage

  • localStorage - дані зберігаються безстроково, до ~5-10 МБ
  • sessionStorage - дані зникають при закритті вкладки
  • Дані доступні тільки на клієнті (не передаються на сервер автоматично)
  • Можна зберігати тільки рядки (для об'єктів потрібен JSON)
  • Синхронний API (блокує виконання коду)

Основи localStorage

// Збереження даних
localStorage.setItem('username', 'Іван');
localStorage.setItem('age', '25');
localStorage.setItem('isLoggedIn', 'true');

// Отримання даних
const username = localStorage.getItem('username');
console.log(username);  // 'Іван'

// Видалення одного елемента
localStorage.removeItem('age');

// Очищення всього сховища
localStorage.clear();

// Перевірка наявності ключа
if (localStorage.getItem('username') !== null) {
    console.log('Користувач знайдений');
}

// Альтернативний синтаксис (не рекомендується)
localStorage.username = 'Петро';
const name = localStorage.username;
delete localStorage.username;

// Кількість збережених елементів
const itemCount = localStorage.length;

// Отримання ключа за індексом
for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    const value = localStorage.getItem(key);
    console.log(key, value);
}

// Перебір всіх елементів
for (let key in localStorage) {
    if (localStorage.hasOwnProperty(key)) {
        console.log(key, localStorage.getItem(key));
    }
}

// sessionStorage працює ідентично
sessionStorage.setItem('tempData', 'тимчасові дані');
const tempData = sessionStorage.getItem('tempData');

JSON - серіалізація та десеріалізація

// JSON (JavaScript Object Notation) - формат для обміну даними

// Об'єкт JavaScript
const user = {
    name: 'Іван',
    age: 25,
    city: 'Київ',
    hobbies: ['читання', 'програмування', 'спорт'],
    isActive: true
};

// JSON.stringify - конвертує об'єкт в рядок JSON
const jsonString = JSON.stringify(user);
console.log(jsonString);
// '{"name":"Іван","age":25,"city":"Київ",...}'

// З форматуванням (красивий вивід)
const prettyJson = JSON.stringify(user, null, 2);
console.log(prettyJson);

// JSON.parse - конвертує рядок JSON в об'єкт
const parsedUser = JSON.parse(jsonString);
console.log(parsedUser.name);  // 'Іван'

// Збереження об'єкта в localStorage
const settings = {
    theme: 'dark',
    language: 'uk',
    notifications: true
};

localStorage.setItem('settings', JSON.stringify(settings));

// Отримання об'єкта з localStorage
const savedSettings = JSON.parse(localStorage.getItem('settings'));
console.log(savedSettings.theme);  // 'dark'

// Збереження масиву
const todos = [
    { id: 1, text: 'Купити молоко', done: false },
    { id: 2, text: 'Вивчити JavaScript', done: true }
];

localStorage.setItem('todos', JSON.stringify(todos));

// Отримання масиву
const savedTodos = JSON.parse(localStorage.getItem('todos'));
console.log(savedTodos[0].text);  // 'Купити молоко'

// Обробка помилок при парсингу
try {
    const data = JSON.parse(localStorage.getItem('someKey'));
} catch (error) {
    console.error('Помилка парсингу JSON:', error);
    // Використати значення за замовчуванням
    const data = { default: true };
}

// Безпечне отримання з перевіркою
function getFromStorage(key, defaultValue) {
    try {
        const item = localStorage.getItem(key);
        return item ? JSON.parse(item) : defaultValue;
    } catch (error) {
        return defaultValue;
    }
}

CRUD операції з localStorage

// Create - створення/додавання
function addTodo(todo) {
    const todos = getTodos();
    todos.push({
        id: Date.now(),
        text: todo,
        done: false,
        createdAt: new Date().toISOString()
    });
    saveTodos(todos);
}

// Read - читання
function getTodos() {
    const todos = localStorage.getItem('todos');
    return todos ? JSON.parse(todos) : [];
}

function getTodoById(id) {
    const todos = getTodos();
    return todos.find(todo => todo.id === id);
}

// Update - оновлення
function updateTodo(id, updates) {
    const todos = getTodos();
    const index = todos.findIndex(todo => todo.id === id);
    
    if (index !== -1) {
        todos[index] = { ...todos[index], ...updates };
        saveTodos(todos);
        return true;
    }
    return false;
}

function toggleTodo(id) {
    const todos = getTodos();
    const todo = todos.find(t => t.id === id);
    
    if (todo) {
        todo.done = !todo.done;
        saveTodos(todos);
    }
}

// Delete - видалення
function deleteTodo(id) {
    const todos = getTodos();
    const filtered = todos.filter(todo => todo.id !== id);
    saveTodos(filtered);
}

function deleteAllTodos() {
    localStorage.removeItem('todos');
}

// Helper функція для збереження
function saveTodos(todos) {
    localStorage.setItem('todos', JSON.stringify(todos));
}

// Приклад використання
addTodo('Вивчити localStorage');
addTodo('Створити TODO додаток');

const allTodos = getTodos();
console.log('Всі завдання:', allTodos);

updateTodo(allTodos[0].id, { text: 'Освоїти localStorage' });
toggleTodo(allTodos[0].id);
deleteTodo(allTodos[1].id);

// Пошук та фільтрація
function searchTodos(query) {
    const todos = getTodos();
    return todos.filter(todo => 
        todo.text.toLowerCase().includes(query.toLowerCase())
    );
}

function getCompletedTodos() {
    const todos = getTodos();
    return todos.filter(todo => todo.done);
}

function getActiveTodos() {
    const todos = getTodos();
    return todos.filter(todo => !todo.done);
}

Відстеження змін у Storage

// Подія storage спрацьовує при зміні localStorage в іншій вкладці/вікні
window.addEventListener('storage', (e) => {
    console.log('Storage змінено:');
    console.log('Ключ:', e.key);
    console.log('Старе значення:', e.oldValue);
    console.log('Нове значення:', e.newValue);
    console.log('URL:', e.url);
    console.log('Storage:', e.storageArea);
    
    // Синхронізація даних між вкладками
    if (e.key === 'todos') {
        loadTodos();
        renderTodos();
    }
});

// Власний обробник змін для поточної вкладки
const originalSetItem = localStorage.setItem;

localStorage.setItem = function(key, value) {
    const oldValue = localStorage.getItem(key);
    originalSetItem.apply(this, arguments);
    
    // Викликати власний обробник
    const event = new CustomEvent('localStorageChanged', {
        detail: { key, oldValue, newValue: value }
    });
    window.dispatchEvent(event);
};

// Використання власної події
window.addEventListener('localStorageChanged', (e) => {
    console.log('localStorage змінено локально:', e.detail);
});

// Приклад: синхронізація теми між вкладками
window.addEventListener('storage', (e) => {
    if (e.key === 'theme') {
        applyTheme(e.newValue);
    }
});

function applyTheme(theme) {
    document.body.className = theme;
}

// Зберегти тему
function setTheme(theme) {
    localStorage.setItem('theme', theme);
    applyTheme(theme);
}

// 2. Кошик покупок
const cart = {
    add(product) {
        const items = this.getItems();
        items.push(product);
        localStorage.setItem('cart', JSON.stringify(items));
    },
    
    getItems() {
        return JSON.parse(localStorage.getItem('cart') || '[]');
    },
    
    remove(productId) {
        const items = this.getItems().filter(item => item.id !== productId);
        localStorage.setItem('cart', JSON.stringify(items));
    },
    
    clear() {
        localStorage.removeItem('cart');
    },
    
    getTotal() {
        return this.getItems().reduce((sum, item) => sum + item.price, 0);
    }
};

// 3. Історія пошуку
function addSearchQuery(query) {
    const history = JSON.parse(localStorage.getItem('searchHistory') || '[]');
    
    // Видалити дублікат якщо є
    const filtered = history.filter(q => q !== query);
    
    // Додати на початок
    filtered.unshift(query);
    
    // Обмежити до 10 останніх
    const limited = filtered.slice(0, 10);
    
    localStorage.setItem('searchHistory', JSON.stringify(limited));
}

// 4. Збереження стану форми
function saveFormState(formId) {
    const form = document.querySelector(formId);
    const data = {};
    
    new FormData(form).forEach((value, key) => {
        data[key] = value;
    });
    
    localStorage.setItem(`form_${formId}`, JSON.stringify(data));
}

function restoreFormState(formId) {
    const saved = localStorage.getItem(`form_${formId}`);
    if (!saved) return;
    
    const data = JSON.parse(saved);
    const form = document.querySelector(formId);
    
    Object.keys(data).forEach(key => {
        const element = form.elements[key];
        if (element) element.value = data[key];
    });
}

// 5. Таймер/Секундомір з збереженням стану
class Timer {
    constructor() {
        this.loadState();
    }
    
    start() {
        this.startTime = Date.now();
        this.isRunning = true;
        this.saveState();
    }
    
    stop() {
        this.elapsed = this.getElapsed();
        this.isRunning = false;
        this.saveState();
    }
    
    getElapsed() {
        if (this.isRunning) {
            return this.elapsed + (Date.now() - this.startTime);
        }
        return this.elapsed;
    }
    
    saveState() {
        localStorage.setItem('timerState', JSON.stringify({
            elapsed: this.elapsed,
            isRunning: this.isRunning,
            startTime: this.startTime
        }));
    }
    
    loadState() {
        const saved = localStorage.getItem('timerState');
        const state = saved ? JSON.parse(saved) : {
            elapsed: 0,
            isRunning: false
        };
        
        this.elapsed = state.elapsed;
        this.isRunning = state.isRunning;
        this.startTime = state.startTime;
        
        if (this.isRunning) {
            this.startTime = Date.now() - this.elapsed;
        }
    }
}
