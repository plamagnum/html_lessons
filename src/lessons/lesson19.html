Урок 19: Fetch API та асинхронність

Promises, async/await, HTTP запити, робота з API

Що таке асинхронність?

Асинхронне програмування дозволяє виконувати тривалі операції (запити до сервера, читання файлів) без блокування виконання коду. JavaScript використовує Promise та async/await для роботи з асинхронними операціями.

Основні концепції

  • Promise - об'єкт, що представляє результат асинхронної операції
  • async/await - синтаксичний цукор для роботи з Promise
  • Fetch API - сучасний спосіб виконання HTTP запитів
  • JSON API - формат обміну даними з сервером
  • Обробка помилок через try/catch

Promises - основи

// Promise має три стани: pending, fulfilled, rejected

// Створення Promise
const promise = new Promise((resolve, reject) => {
    // Асинхронна операція
    setTimeout(() => {
        const success = true;
        
        if (success) {
            resolve('Операція успішна!');  // виконано успішно
        } else {
            reject('Помилка!');  // відхилено з помилкою
        }
    }, 1000);
});

// Обробка результату Promise
promise
    .then(result => {
        console.log(result);  // 'Операція успішна!'
    })
    .catch(error => {
        console.error(error);
    })
    .finally(() => {
        console.log('Завершено (незалежно від результату)');
    });

// Ланцюжок Promise
fetch('https://api.example.com/data')
    .then(response => response.json())
    .then(data => {
        console.log(data);
        return data.userId;
    })
    .then(userId => {
        return fetch(`https://api.example.com/users/${userId}`);
    })
    .then(response => response.json())
    .then(user => {
        console.log(user);
    })
    .catch(error => {
        console.error('Помилка:', error);
    });

// Promise.all - чекає виконання всіх Promise
const promise1 = fetch('/api/users');
const promise2 = fetch('/api/posts');
const promise3 = fetch('/api/comments');

Promise.all([promise1, promise2, promise3])
    .then(responses => {
        return Promise.all(responses.map(r => r.json()));
    })
    .then(([users, posts, comments]) => {
        console.log('Користувачі:', users);
        console.log('Пости:', posts);
        console.log('Коментарі:', comments);
    });

// Promise.race - повертає результат першого виконаного Promise
const timeout = new Promise((_, reject) => 
    setTimeout(() => reject('Timeout'), 5000)
);

Promise.race([fetch('/api/data'), timeout])
    .then(response => response.json())
    .catch(error => console.error(error));

// Promise.allSettled - чекає виконання всіх (успішних і неуспішних)
Promise.allSettled([promise1, promise2, promise3])
    .then(results => {
        results.forEach(result => {
            if (result.status === 'fulfilled') {
                console.log('Успіх:', result.value);
            } else {
                console.log('Помилка:', result.reason);
            }
        });
    });

Async/Await - сучасний підхід

// async функція завжди повертає Promise
async function fetchData() {
    return 'Дані';
}

fetchData().then(result => console.log(result));

// await чекає виконання Promise (можна використовувати тільки в async функціях)
async function loadUserData() {
    try {
        const response = await fetch('https://api.example.com/user');
        const user = await response.json();
        console.log(user);
        return user;
    } catch (error) {
        console.error('Помилка завантаження:', error);
    }
}

// Приклад з декількома запитами
async function loadData() {
    try {
        // Послідовні запити
        const user = await fetch('/api/user').then(r => r.json());
        const posts = await fetch(`/api/posts?userId=${user.id}`).then(r => r.json());
        const comments = await fetch(`/api/comments?postId=${posts[0].id}`).then(r => r.json());
        
        return { user, posts, comments };
    } catch (error) {
        console.error('Помилка:', error);
    }
}

// Паралельні запити (швидше)
async function loadDataParallel() {
    try {
        const [user, posts, comments] = await Promise.all([
            fetch('/api/user').then(r => r.json()),
            fetch('/api/posts').then(r => r.json()),
            fetch('/api/comments').then(r => r.json())
        ]);
        
        return { user, posts, comments };
    } catch (error) {
        console.error('Помилка:', error);
    }
}

// Обробка помилок
async function fetchWithErrorHandling() {
    try {
        const response = await fetch('/api/data');
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const data = await response.json();
        return data;
    } catch (error) {
        console.error('Помилка:', error);
        throw error;  // передати помилку далі
    } finally {
        console.log('Завершено');
    }
}

// Використання async/await з циклами
async function processItems(items) {
    // Послідовна обробка
    for (const item of items) {
        const result = await processItem(item);
        console.log(result);
    }
    
    // Паралельна обробка
    const results = await Promise.all(items.map(item => processItem(item)));
    console.log(results);
}

Fetch API - основи

// Базовий GET запит
fetch('https://api.example.com/data')
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error('Помилка:', error));

// З async/await
async function fetchData() {
    const response = await fetch('https://api.example.com/data');
    const data = await response.json();
    return data;
}

// POST запит з даними
async function createPost(postData) {
    const response = await fetch('https://api.example.com/posts', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
        },
        body: JSON.stringify(postData)
    });
    
    const data = await response.json();
    return data;
}

// PUT запит (оновлення)
async function updatePost(id, postData) {
    const response = await fetch(`https://api.example.com/posts/${id}`, {
        method: 'PUT',
        headers: {
            'Content-Type': 'application/json',
        },
        body: JSON.stringify(postData)
    });
    
    return await response.json();
}

// DELETE запит
async function deletePost(id) {
    const response = await fetch(`https://api.example.com/posts/${id}`, {
        method: 'DELETE'
    });
    
    return response.ok;
}

// Обробка відповіді
async function fetchWithHandling() {
    const response = await fetch('/api/data');
    
    // Перевірка статусу
    if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
    }
    
    // Перевірка типу контенту
    const contentType = response.headers.get('content-type');
    if (contentType && contentType.includes('application/json')) {
        return await response.json();
    } else {
        return await response.text();
    }
}

// Headers - робота з заголовками
const headers = new Headers();
headers.append('Authorization', 'Bearer token123');
headers.append('X-Custom-Header', 'value');

fetch('/api/data', { headers });

// AbortController - скасування запиту
const controller = new AbortController();
const signal = controller.signal;

fetch('/api/data', { signal })
    .then(response => response.json())
    .catch(error => {
        if (error.name === 'AbortError') {
            console.log('Запит скасовано');
        }
    });

// Скасувати через 5 секунд
setTimeout(() => controller.abort(), 5000);

// Timeout для запиту
async function fetchWithTimeout(url, timeout = 5000) {
    const controller = new AbortController();
    const id = setTimeout(() => controller.abort(), timeout);
    
    try {
        const response = await fetch(url, { signal: controller.signal });
        clearTimeout(id);
        return await response.json();
    } catch (error) {
        clearTimeout(id);
        throw error;
    }
}

// Приклад: API погоди
async function getWeather(city) {
    const apiKey = 'YOUR_API_KEY';
    const url = `https://api.openweathermap.org/data/2.5/weather?q=${city}&appid=${apiKey}&units=metric&lang=uk`;
    
    try {
        const response = await fetch(url);
        const data = await response.json();
        
        return {
            temp: data.main.temp,
            description: data.weather[0].description,
            humidity: data.main.humidity
        };
    } catch (error) {
        console.error('Помилка отримання погоди:', error);
    }
}

// Пагінація результатів
async function fetchAllPages(baseUrl) {
    const allData = [];
    let page = 1;
    let hasMore = true;
    
    while (hasMore) {
        const response = await fetch(`${baseUrl}?page=${page}`);
        const data = await response.json();
        
        allData.push(...data);
        
        hasMore = data.length > 0;
        page++;
    }
    
    return allData;
}

// Кешування запитів
const cache = new Map();

async function fetchWithCache(url, ttl = 60000) {
    const cached = cache.get(url);
    
    if (cached && Date.now() - cached.timestamp < ttl) {
        console.log('Використано кеш');
        return cached.data;
    }
    
    const response = await fetch(url);
    const data = await response.json();
    
    cache.set(url, {
        data,
        timestamp: Date.now()
    });
    
    return data;
}

Обробка помилок та індикатори завантаження

// Комплексна обробка помилок
async function fetchDataWithHandling(url) {
    let response;
    
    try {
        response = await fetch(url);
        
        // Перевірка HTTP статусу
        if (!response.ok) {
            const errorData = await response.json().catch(() => ({}));
            throw new Error(errorData.message || `HTTP error! status: ${response.status}`);
        }
        
        // Перевірка Content-Type
        const contentType = response.headers.get('content-type');
        if (!contentType || !contentType.includes('application/json')) {
            throw new Error('Відповідь не у форматі JSON');
        }
        
        const data = await response.json();
        return data;
        
    } catch (error) {
        // Різні типи помилок
        if (error.name === 'TypeError') {
            console.error('Помилка мережі або CORS');
        } else if (error.name === 'AbortError') {
            console.error('Запит скасовано');
        } else {
            console.error('Помилка:', error.message);
        }
        
        throw error;
    }
}

// Loading state manager
class LoadingManager {
    constructor(element) {
        this.element = element;
        this.loading = false;
    }
    
    start() {
        this.loading = true;
        this.element.classList.add('loading');
        this.element.disabled = true;
        this.originalText = this.element.textContent;
        this.element.textContent = 'Завантаження...';
    }
    
    stop() {
        this.loading = false;
        this.element.classList.remove('loading');
        this.element.disabled = false;
        this.element.textContent = this.originalText;
    }
}

// Retry логіка
async function fetchWithRetry(url, options = {}, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
        try {
            const response = await fetch(url, options);
            
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }
            
            return await response.json();
        } catch (error) {
            if (i === maxRetries - 1) {
                throw error;
            }
            
            console.log(`Спроба ${i + 1} не вдалася, повтор через 1 сек...`);
            await new Promise(resolve => setTimeout(resolve, 1000));
        }
    }
}

// Progress indicator
async function downloadWithProgress(url) {
    const response = await fetch(url);
    const contentLength = response.headers.get('content-length');
    const total = parseInt(contentLength, 10);
    let loaded = 0;
    
    const reader = response.body.getReader();
    const chunks = [];
    
    while (true) {
        const { done, value } = await reader.read();
        
        if (done) break;
        
        chunks.push(value);
        loaded += value.length;
        
        const progress = (loaded / total) * 100;
        console.log(`Завантажено: ${progress.toFixed(2)}%`);
    }
    
    const blob = new Blob(chunks);
    return blob;
}
