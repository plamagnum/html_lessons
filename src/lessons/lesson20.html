Урок 20: Фінальний проєкт - TODO додаток

Комплексний проєкт із використанням всіх вивчених технологій

Про проєкт

Цей фінальний проєкт об'єднує всі знання, отримані в попередніх уроках. Ми створимо повноцінний TODO додаток з можливістю додавання, редагування, видалення завдань, фільтрацією, пошуком та збереженням даних у localStorage.

Використані технології та концепції

  • HTML5 - семантична розмітка, форми, атрибути
  • CSS - Flexbox, Grid, анімації, адаптивність
  • JavaScript - змінні, функції, масиви, об'єкти
  • DOM - createElement, querySelector, classList
  • Події - addEventListener, делегування, form submit
  • localStorage - збереження даних, JSON серіалізація
  • Додатково - валідація, фільтрація, пошук, статистика

Повний код TODO додатка

Нижче представлений повний робочий код TODO додатка. Цей код демонструє всі ключові концепції, які ми вивчили протягом курсу. Ви можете запустити його та вивчити, як працює кожна частина.

Запустіть повний TODO додаток:

Розбір ключових моментів

// 1. Структура даних (Об'єкти, Масиви)
const todos = [
    { id: 1, text: 'Завдання', completed: false, priority: 'high' }
];

// 2. DOM маніпуляції (createElement, querySelector)
const todoItem = document.createElement('li');
todoItem.className = 'todo-item';
list.appendChild(todoItem);

// 3. Події (addEventListener, делегування)
list.addEventListener('click', (e) => {
    if (e.target.classList.contains('delete-btn')) {
        deleteTodo(id);
    }
});

// 4. LocalStorage (JSON, збереження)
function saveTodos() {
    localStorage.setItem('todos', JSON.stringify(todos));
}

function loadTodos() {
    return JSON.parse(localStorage.getItem('todos') || '[]');
}

// 5. Фільтрація та пошук (масиви)
function getFilteredTodos() {
    return todos.filter(t => {
        const matchesFilter = filter === 'all' || 
                            (filter === 'active' && !t.completed) ||
                            (filter === 'completed' && t.completed);
        
        const matchesSearch = t.text.toLowerCase().includes(searchQuery);
        
        return matchesFilter && matchesSearch;
    });
}

// 6. Оновлення UI (render функція)
function render() {
    const filtered = getFilteredTodos();
    list.innerHTML = filtered.map(todo => renderTodo(todo)).join('');
    updateStats();
}

// 7. Валідація форми
form.addEventListener('submit', (e) => {
    e.preventDefault();
    
    const input = form.elements.text;
    if (input.value.trim().length < 3) {
        showError('Мінімум 3 символи');
        return;
    }
    
    addTodo(input.value.trim());
    input.value = '';
});

// 8. Обчислення статистики
function getStats() {
    const total = todos.length;
    const completed = todos.filter(t => t.completed).length;
    const active = total - completed;
    const percentage = total > 0 ? (completed / total) * 100 : 0;
    
    return { total, completed, active, percentage };
}

Можливості для розширення

Ідеї для покращення додатка

  • Категорії - додайте категорії завдань (робота, особисте, покупки)
  • Терміни виконання - встановлення дедлайнів для завдань
  • Підзавдання - можливість створювати підзавдання
  • Drag & Drop - перетягування для зміни порядку
  • Теми оформлення - світла/темна тема
  • Експорт/Імпорт - збереження у файл JSON
  • Сортування - за датою, пріоритетом, алфавітом
  • Нотифікації - нагадування про завдання
  • Синхронізація - збереження на сервері через API
  • Статистика - графіки продуктивності

function exportTodos() {
    const dataStr = JSON.stringify(todos, null, 2);
    const dataUri = 'data:application/json;charset=utf-8,' + 
                    encodeURIComponent(dataStr);
    
    const link = document.createElement('a');
    link.setAttribute('href', dataUri);
    link.setAttribute('download', 'todos.json');
    link.click();
}

// Завдання 6: Імпорт даних
// Додайте можливість завантажити файл з завданнями
function importTodos(file) {
    const reader = new FileReader();
    reader.onload = (e) => {
        try {
            const imported = JSON.parse(e.target.result);
            todos = [...todos, ...imported];
            saveTodos();
            render();
        } catch (error) {
            alert('Помилка імпорту файлу');
        }
    };
    reader.readAsText(file);
}

                  ? 'dark' : 'light';
    localStorage.setItem('theme', theme);

function getCompletedToday() {
    const today = new Date().toDateString();
    return todos.filter(t => {
        if (!t.completedAt) return false;
        return new Date(t.completedAt).toDateString() === today;
    }).length;
}
